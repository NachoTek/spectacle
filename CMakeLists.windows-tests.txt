cmake_minimum_required(VERSION 3.16)
project(SpectacleWindowsTests LANGUAGES CXX)

set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)

# Enable Qt automoc for Q_OBJECT macro processing
set(CMAKE_AUTOMOC ON)
set(CMAKE_AUTORCC ON)
set(CMAKE_AUTOUIC ON)

# Find Qt 6 - add Gui component for QScreen, plus Qml, Quick, and Widgets for post-capture viewer
find_package(Qt6 REQUIRED COMPONENTS Core Gui Qml Quick Widgets Test)

# Include Qt test framework
enable_testing()
include(CTest)

# Parent directory (where source files are)
set(SOURCE_DIR ${CMAKE_CURRENT_SOURCE_DIR}/..)

# Set up include directories
include_directories(${SOURCE_DIR}/src)

# Windows Platform Components - Helper Process
set(HELPER_PROCESS_SOURCES
    ${SOURCE_DIR}/src/Platforms/Windows/HelperProcess.cpp
    ${SOURCE_DIR}/src/Platforms/Windows/IPCProtocol.cpp
)

# Windows Platform Components - Named Pipe Server
set(NAMED_PIPE_SERVER_SOURCES
    ${SOURCE_DIR}/src/Platforms/Windows/NamedPipeServer.cpp
    ${SOURCE_DIR}/src/Platforms/Windows/IPCProtocol.cpp
)

# Windows Platform Components - WGC Capture
set(WGC_CAPTURE_SOURCES
    ${SOURCE_DIR}/src/Platforms/Windows/WGCCapture.cpp
)

# Geometry - Selection
set(GEOMETRY_SOURCES
    ${SOURCE_DIR}/src/Gui/Overlay/SelectionGeometry.cpp
)

# Windows Platform Components - Window Event Monitor (Story 1.8)
set(WINDOW_EVENT_MONITOR_SOURCES
    ${SOURCE_DIR}/src/Platforms/Windows/WindowEventMonitor.cpp
)

# Create test executables
add_executable(helper_process_test
    ${SOURCE_DIR}/tests/Integration/HelperProcessTest.cpp
    ${HELPER_PROCESS_SOURCES}
)
target_link_libraries(helper_process_test Qt6::Core Qt6::Test)

add_executable(named_pipe_server_test
    ${SOURCE_DIR}/tests/Integration/NamedPipeServerTest.cpp
    ${NAMED_PIPE_SERVER_SOURCES}
)
target_link_libraries(named_pipe_server_test Qt6::Core Qt6::Test)

add_executable(selection_geometry_test
    ${SOURCE_DIR}/tests/Unit/SelectionGeometryTest.cpp
    ${GEOMETRY_SOURCES}
)
target_link_libraries(selection_geometry_test Qt6::Core Qt6::Gui Qt6::Test)

# Window Event Monitor Test (Story 1.8)
add_executable(window_event_monitor_test
    ${SOURCE_DIR}/tests/Integration/WindowEventMonitorTest.cpp
    ${WINDOW_EVENT_MONITOR_SOURCES}
)
target_link_libraries(window_event_monitor_test Qt6::Core Qt6::Gui Qt6::Test)

# Annotation Data Model Test (Story 1.3)
set(ANNOTATION_SOURCES
    ${SOURCE_DIR}/src/Gui/Annotation/Annotation.cpp
    ${SOURCE_DIR}/src/Gui/Annotation/AnnotationListModel.cpp
    ${SOURCE_DIR}/src/Gui/Annotation/AnnotationRenderer.cpp
)

add_executable(annotation_test
    ${SOURCE_DIR}/tests/Unit/AnnotationTest.cpp
    ${ANNOTATION_SOURCES}
)
target_link_libraries(annotation_test Qt6::Core Qt6::Gui Qt6::Test)

# Post-Capture Annotation Viewer Test (Story 1.4)
set(POSTCAPTURE_SOURCES
    ${SOURCE_DIR}/src/Gui/Annotation/Annotation.cpp
    ${SOURCE_DIR}/src/Gui/Annotation/AnnotationListModel.cpp
    ${SOURCE_DIR}/src/Gui/Annotation/AnnotationRenderer.cpp
    ${SOURCE_DIR}/src/Gui/Annotation/CapturedImageProvider.cpp  # CRITICAL #3
    ${SOURCE_DIR}/src/Gui/Annotation/PostCaptureAnnotationViewer.cpp
)

add_executable(postcapture_test
    ${SOURCE_DIR}/tests/Unit/PostCaptureAnnotationViewerTest.cpp
    ${POSTCAPTURE_SOURCES}
)
target_link_libraries(postcapture_test Qt6::Core Qt6::Gui Qt6::Qml Qt6::Quick Qt6::Widgets Qt6::Test)

# Add tests to CTest
add_test(NAME HelperProcessTest COMMAND helper_process_test)
add_test(NAME NamedPipeServerTest COMMAND named_pipe_server_test)
add_test(NAME SelectionGeometryTest COMMAND selection_geometry_test)
add_test(NAME WindowEventMonitorTest COMMAND window_event_monitor_test)
add_test(NAME AnnotationTest COMMAND annotation_test)
add_test(NAME PostCaptureAnnotationViewerTest COMMAND postcapture_test)
